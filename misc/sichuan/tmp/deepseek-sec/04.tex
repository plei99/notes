\subsection{预稳定曲线上的典范对数结构}%
\label{sub:Canonical log structures on prestable curves}

设 \(\mf{M}_{g,n}\) 为 \(n\)-标记预稳定曲线的模空间。则奇点轨迹是一个正规交叉除子，因此存在一个典范对数结构
\[ M_{\mf{M}_{g,n}}^{\on{can}} = M_{\partial \mf{M}_{g,n}}. \]
若考虑万有族 \(\mf{C}_{g,n}\)，我们可以构造 \(\pi^{-1}(\partial \mf{M}_{g,n}) \subseteq \mf{C}_{g,n}\)。于是，我们得到一个映射
\[ \pi^* M_{\partial \mf{M}_{g,n}} \to M_{\pi^{-1}(\partial \mf{M}_{g,n})}. \]
我们也可以考虑除子 \(p_i(\mf{M}_{g,n}) \subset \mf{C}_{g,n}\)，它们同样定义了典范对数结构
\[ P_i \coloneqq M_{p_i(\mf{M}_{g,n})}. \]
因此，我们可以定义对数结构
\[ M_{\mf{C}_{g,n}}^{\on{can}} \coloneqq M_{\pi^{-1}(\partial\mf{M}_{g,n})} \oplus_{\mc{O}^{\times}} \bigoplus_{i=1}^n P_i. \]
那么若考虑
\[ \pi^{\flat} \colon \ul{\pi}^{-1} M_{\mf{M}_{g,n}}^{\on{can}} \to M_{\pi^{-1}(\partial \mf{M}_{g,n})} \to M_{\mf{C}_{g,n}}^{\on{can}}, \]
我们便可以将 \(\pi\) 提升为对数叠之间的映射。

\begin{thm}[{\cite{FKato2000}}]
    映射 \(\pi\) 是对数光滑的对数积（integral）映射。
\end{thm}

\begin{defn}
    对于预稳定曲线的族 \((\ul{C}/\ul{S}, \vec{p})\)，我们可以沿图表
    \begin{equation*}
    \begin{tikzcd}
        \ul{C} \ar{r} \ar{d} & \mf{C}_{g,n} \ar{d} \\
        S \ar{r} & \mf{M}_{g,n}
    \end{tikzcd}
    \end{equation*}
    作拉回，并为左侧边赋予一个典范对数结构。
\end{defn}

\begin{rmk}
    设 \(\ul{S} = \Spec k\)。若 \(S^{\on{can}} = \Spec (\N^r \to k)\)，则 \(r\) 是 \(\ul{C}\) 上结点的个数。在结点 \(q\) 上 \'etale 局部地，对数结构形如
    \begin{equation*}
    \begin{tikzcd}
        \N^{r-1} \oplus \N^2 \ar{r} & k[x,y]/xy \\
        \N^{r-1} \oplus \N \ar{u}{\on{id}\oplus\Delta} \ar{r} & k \ar{u}.
    \end{tikzcd}
    \end{equation*}
\end{rmk}

\begin{thm}[{\cite{FKato2000}}]
    设 \((\mc{C}/S, \vec{p})\) 是 fs 基 \(S\) 上的 \(n\)-标记对数曲线。则存在唯一的 \(M_S^{\on{can}} \to M_S\)，使得
    \begin{equation*}
    \begin{tikzcd}
        \mc{C} \ar{d} \ar{r} & \mc{C}^{\on{can}} \ar{d} \\
        S \ar{r}{a} & S^{\on{can}}
    \end{tikzcd}
    \end{equation*}
    是 fs 对数概形范畴中的笛卡儿图表。
\end{thm}

一句口号是：对数曲线就是预稳定曲线连同一个 \(M_S^{\on{can}} \to M_S\)。

\begin{exm}
    设 \(\ul{S} = \Spec k[s]\) 及 \(\ul{T} = \Spec k[t]\)。再取 \(\ell \in \N\)。则考虑图表
    \begin{equation*}
    \begin{tikzcd}
        \Spec \frac{k{ [x,y,s] }}{xy-s^{\ell}} \ar{d} \ar{r} & \Spec \frac{k{ [x,y,s] }}{xy-s^{\ell}} \ar{d} \ar{r} & \Spec  \frac{k{[x,y,t]}}{xy-t} \ar{r}\ar{d} & \mf{C}_{g,n} \ar{d} \\
        \Spec k{[s]} \ar{r}{\ul{ \on{id} }} & \Spec {k[s]} \ar{r} & \Spec k{[t]} \ar{r} & \mf{M}_{g,n}.
    \end{tikzcd}
    \end{equation*}
    在对数结构上，考虑如下情形：左下角的箭头在 \(\N\) 上对应 \(1 \mapsto \ell\)。则在左上角，幺半群变为 \(\N^2 \oplus_{\N, \ell} \N\)。若将其记作 \(P\)，则映射 \(k[P] \to k[\N]\) 并非带有典范对数结构的对数曲线族，而是一个对数曲线族。
\end{exm}

\subsection{对数曲线的模空间}%
\label{sub:Moduli of log curves}

\begin{defn}
    \(n\)-标记对数曲线之间的映射 \((\mc{C}, S) \to (\mc{C}', S')\) 是一对 \((\ell \colon \mc{C} \to \mc{C}', \Theta \colon S \to S')\)，使得 \(\Theta\) 是严格的（strict），且图表
    \begin{equation*}
    \begin{tikzcd}
        \mc{C} \ar{d} \ar{r}{\ell} & \mc{C}' \ar{d} \\
        S \ar{r}{\Theta} & S'
    \end{tikzcd}
    \end{equation*}
    是笛卡儿图表。记 \(\mf{M}_{g,n}^{\log}\) 为参数化 \(n\)-标记对数曲线的纤维范畴。
\end{defn}

\begin{prop}[{\cite{FKato2000,Olsson2003}}]
    我们有 \(\mf{M}_{g,n}^{\log} \simeq \ms{Log}_{(\mf{M}_{g,n}, M_{\mf{M}_{g,n}}^{\on{can}})}\)。
\end{prop}

特别地，对于概形 \(\ul{T}\)，群胚 \(\mf{M}_{g,n}^{\log}(\ul{T})\) 参数化对数结构 \(M_T\) 连同对数曲线 \(\mc{C} / T\)。

考虑 \(S = \Spec (Q \to k)\) 上的对数曲线 \(\pi \colon \mc{C} \to S, \vec{p}\)。记 \(\bar{\pi}^{\flat} \colon \bar{\pi}^* \bar{M}_S \to \bar{M}_{\mc{C}}\)。在一般点或一般光滑点 \(\eta\) 处，该映射为
\[ \bar{\pi}_{\eta}^{\flat} \colon (\pi^{-1} \bar{M}_S)_{\eta} = Q \simeq \bar{M}_{\mc{C}, \eta}. \]
在标记点 \(p_i \in \ol{\ab\{\eta\}}\) 处，我们有 \(\bar{M}_{\mc{C}, p_i} = Q \oplus \N\)。此时我们有特殊化映射
\[ \chi_{\eta} \colon \bar{M}_{\mc{C}, p_i} \xrightarrow{\on{pr}_1} \bar{M}_{\mc{C}, \eta} \]
以及拉回
\[ \bar{\pi}_{p_i}^{\flat} \colon Q \hookrightarrow Q \oplus \N. \]
在结点 \(q\) 处，我们有 \(\bar{M}_{\mc{C}, q} = Q \oplus_{\N} \N^2\)。设 \(\rho_q\) 记 \(1\) 在 \(Q\) 中的像，即 \(q\) 处的边长度或光滑化参数。记 \(\bar{\eta}_1\) 与 \(\bar{\eta}_2\) 为 \(q\) 的两个分支的一般点。则广义化映射为
\begin{align*}
    \chi_{\eta_1} \colon Q \oplus_{\N} \N^2 \to Q \qquad (q, (a,b)) \mapsto q + a \rho_q ;\\
    \chi_{\eta_2} \colon Q \oplus_{\N} \N^2 \to Q \qquad (q, (a,b)) \mapsto q + b \rho_q .
\end{align*}

\subsection{对数曲线的热带化}%
\label{sub:Tropicalization of log curves}

我们只考虑对数点 \(S = \Spec(Q \to k)\) 上的情形。我们将考虑锥范畴 \(\ms{Cone}\)，其对象为满足 \(w \subseteq N_w \otimes_{\Z} \R\) 的数对 \((w, N_w)\)，其中 \(w\) 是一个满维强凸有理多面体锥。态射为映射 \((w_1, N_{w_1}) \to (w_2, N_{w_2})\)，它由
\[ \chi \colon N_{w_1} \to N_{w_2} \]
给出，并满足 \(\chi_{\R}(w_1) \subseteq w_2\)。

\begin{exm}
    设 \(X\) 为 fs 对数概形，考虑 \(( (\bar{M}_{X,x}^{\vee})_{\R}, (\bar{M}_{X,x}^{\vee})_{\Z} )\)。
\end{exm}

我们现在考虑广义锥复形范畴 \(\ms{GCC}\)。其对象是拓扑空间 \(\abs{\Sigma}\) 连同一个将其表示为 \(\ms{Cone}\) 中某图表的余极限的表述，其中所有箭头都是面的包含。其态射是连续映射 \(\abs{\Sigma} \to \abs{\Sigma'}\)，使得对每个 \((w, N_w) \in \Sigma\)，映射 \(w \to \abs{\Sigma'}\) 分解通过某个 \((w, N_w) \to (w', N_{w'})\)。

\begin{exm}
    若 \(f \colon X \to Y\) 是 fs 对数概形之间的态射，则映射 \(\bar{f}^{\flat} \colon f^* \bar{M}_Y \to \bar{M}_X\) 在热带化上诱导出 \(\Sigma(X) \to \Sigma(Y)\)。
\end{exm}

\begin{defn}
    一个\textit{锥复形}是一个广义锥复形 \(\Sigma\)，使得对每个 \((\sigma, N_{\sigma}) \in \Sigma\)，映射 \(\sigma \to \abs{\Sigma}\) 都是单射。
\end{defn}

\begin{thm}[{\cite{ACP2015}}]
    若 \(X\) 是 fs 的，且在带有 Zariski 对数结构的 \(\ul{\C}\) 上对数光滑，则 \(\Sigma(X)\) 是一个锥复形。
\end{thm}

对于幺半群 \(Q\)，我们记 \(w_Q = Q_{\R}^{\vee} = \Hom(Q, \R_{\geq 0})\)，并记格 \(\Hom(Q, \Z) = Q_{\Z}^{\vee}\)。在光滑点处，我们有
\[ w_{\eta} = (Q_{\R}^{\vee}, Q_{\Z}^{\vee}). \]
在标记点处，我们看到
\[ w_p = \ab(\Hom (Q \oplus \N, \R_{\geq 0}) = Q_{\R}^{\vee} \oplus \R_{\geq 0}, Q_{\Z}^{\vee}\oplus \Z). \]
在结点处，我们有
\begin{align*}
    w_q &= \ab(\Hom(Q \oplus_{\N} \N^2, \R_{\geq 0}), ( \bar{M}_{\mc{C}, q} )_{\Z}^{\vee}).
\end{align*}
注意到该锥实际上等于
\[ \ab\{(m, m_x, m_y) \mid m(\rho_q) = m_x + m_y \} \simeq \ab\{(m, e_x) \mid e_x \leq m(\rho_q) \} \subseteq Q_{\R}^{\vee} \times \R_{\geq 0}. \]

现在我们已经理解了每个点所对应的锥，接下来需要讨论面映射。对应于标记点的广义化映射为
\[ Q \oplus \N \xrightarrow{\on{pr}_1} Q, \]
相应的面映射为
\[ w_{\eta} \hookrightarrow w_p \qquad m \mapsto (m,0). \]
在结点处，两个面映射为
\[ \chi_{\eta_1}^{\vee} \colon Q_{\R}^{\vee} \to w_q \qquad m \mapsto (m, m(\rho_q)) \]
以及
\[ \chi_{\eta_2}^{\vee} \colon Q_{\R}^{\vee} \to w_q \qquad m \mapsto (m, 0). \]

\begin{exm}
    考虑一条由两条有理曲线并成、且在其中一个分支上有一个标记点 \(p\) 的曲线。设 \(\ul{S} = \Spec \C\)。则我们看到
    \[ S^{\on{can}} = \Spec(\N \to \C), \]
    \(Q = \N\)，且 \(Q_{\R}^{\vee} = \R_{\geq 0}\)。我们同时看到 \(\bar{M}_{C, q} = \N \oplus_{\N} \N^2\) 以及 \(\bar{M}_{C, p} = \N \oplus \N\)。
\end{exm}

\subsection{热带曲线}%
\label{sub:Tropical curves}

\begin{defn}
    锥 \((w, N_w)\) 上的一族\textit{热带曲线} \((G, \vec{g}, \vec{\ell})\) 由以下数据组成：
    \begin{enumerate}
        \item 一个连通图 \(G = V(G) \cup E(G) \cup L(G)\)；
        \item 一个腿的排序 \(\vec{m} \colon L(G) \xrightarrow{\sim} \ab\{1, \ldots, n\}\)；
        \item 一个亏格函数 \(\vec{g} \colon V(G) \to \N\)；
        \item 一个边长度函数 \(\ell \colon E(G) \to \Hom(w \cap N_w, \N) \setminus \ab\{0\}\)。
    \end{enumerate}
\end{defn}

现在我们构造 \(\Sigma(G, \ell)\)。对 \(v \in V(G)\)，令 \(w_v = w\)。对 \(e \in E(G)\)，令
\[ w_e = \ab\{ (s, \lambda) \mid \lambda \leq \vec{\ell}(e)(s) \} \subseteq w \times \R_{\geq 0}. \]
对腿 \(\ell \in L(G)\)，令 \(w_{\ell} = w \times \R_{\geq 0}\)。现在需要定义面映射。若 \(\ell\) 关联于 \(v\)，则面映射为
\[ w_v \hookrightarrow w_{\ell} \colon s \mapsto (s, 0). \]
若 \(e\) 关联于两个顶点 \(v_1, v_2\)，则面映射为
\begin{align*}
    w_{v_1} \hookrightarrow w_e &\colon s \mapsto (s, \vec{\ell}(e)(s)) \\
    w_{v_2} \hookrightarrow w_e & \colon s \mapsto (s, 0).
\end{align*}
\begin{rmk}
    \(v_1\) 与 \(v_2\) 的选取在相差同构的意义下不改变锥复形，但当我们想要为对数映射定义接触阶数时，这一选取是有影响的。然而，一般说来并不存在作出这一选取的典范方式。
\end{rmk}

\subsection{穿孔曲线}%
\label{sub:Punctured curves}

穿孔对数曲线与映射在~\cite{ACGS2025} 中得到了发展。

我们考虑一条结点曲线
\[ \Spec k[x,y]/xy \to \Spec k \]
其对数结构由带有图表 \((a,b) \mapsto x^a y^b\) 的 \(\N^2\) 以及作为标准对数点的 \(\N \to k\) 给出。现在考虑 \(\ul{C}_1 = Z(x) = \Spec k[y]\)，并用 \(p \colon \Spec k \to \Spec k[y]\) 标记该结点。注意到
\[ C^{\on{can}} = (\ul{C}, M_C) \to S^{\on{can}} = \Spec(\N \to k). \]
记 \(M_{C_1}^{\circ} = M_C|_{C_1}\)。则我们看到，在对数结构上有图表
\begin{equation*}
\begin{tikzcd}
    k[y] & & \N^2 \ar{ll}[swap]{(a,b) \mapsto 0^a y^b} \\
    k \ar{u} & & \N \ar{ll} \ar{u}{\Delta}.
\end{tikzcd}
\end{equation*}
特别地，我们可以计算茎 \(\bar{M}_{C_1}^{\circ}|_p = \N^2\)，因此我们有
\[ \bar{\pi}_{C_1^{\circ}}^{\flat} |_p \colon \N \xrightarrow{\Delta} \N^2. \]
然而，我们应当有映射
\[ \bar{\pi}^{\flat}_{C_1}|_p \colon \N \xrightarrow{i_1} \N \oplus \N, \]
因此 \(C_1\) \textbf{不}是对数曲线。于是我们可以考虑图表
\begin{equation*}
\begin{tikzcd}
    \N \oplus \Z \\
    \N^2 \ar[hookrightarrow]{u} \ar{rr}{(0,1)\mapsto (0,1)}[swap]{(1,0)\mapsto (1,1)} & & \N^2 \ar{ull}[swap]{\Vectorstack{{(1,0)\mapsto(1,-1)} {(0,1)\mapsto(0,1)}}} \\
    & \N \ar{ul}{i_1} \ar{ur}{\Delta}.
\end{tikzcd}
\end{equation*}
因此，我们将考虑扩展对数结构。

回忆对于一条对数曲线，我们有 \(M_{\mc{C}} = M_{\mc{C}}' \oplus_{\mc{O}_{\mc{C}}^{\times}} P_{\mc{C}}\)，其中 \(P_{\mc{C}}\) 是由标记给出的除子对数结构。

\begin{defn}
    fs 对数概形 \(S\) 上的一条\textit{穿孔曲线}由 \((\mc{C}^{\circ} \xrightarrow{p} \mc{C} \xrightarrow{\pi} S, \{p_i\}_{i=1}^n)\) 组成，其中 \((\mc{C} \xrightarrow{\pi} S, \vec{p})\) 是一条对数曲线，且映射 \(p \colon \mc{C}^{\circ} \to \mc{C}\) 是 fine 对数概形之间的映射，满足如下条件：
    \begin{enumerate}
        \item 底空间映射 \(\ul{p}\) 是同构；
        \item 在对数结构上，映射 \(p^{\flat} \colon M_{\mc{C}} \hookrightarrow M_{\mc{C}^{\circ}}\) 在远离 \(p_i\) 处是同构，并诱导出
            \[ M_{\mc{C}} \xrightarrow{p^{\flat}} M_{\mc{C}^{\circ}} \hookrightarrow M_{\mc{C}} \oplus P^{\mr{gp}}. \]
        \item 对任意标记 \(p_i\) 以及任意 \(s \in M_{\mc{C}^{\circ}, p_i} \setminus M_{\mc{C}, p_i}\)，我们有 \(\alpha_{\mc{C}^{\circ}}(s) = 0\)。
    \end{enumerate}
\end{defn}

粗略地说，一条穿孔曲线就是一条对数曲线再加上一个交换图表的数据
\begin{equation*}
\begin{tikzcd}
    M_{\mc{C}^{\circ}} \ar{dr} \ar[hookrightarrow]{r} & M_{\mc{C}}' \oplus_{\mc{O}_{\mc{C}}^{\times}} P^{\on{gp}} \\
    & \mc{O}_{\mc{C}} \\
    M_{\mc{C}} \ar[hookrightarrow]{uu} \ar{ur}.
\end{tikzcd}
\end{equation*}


\begin{exm}
    设 \((\ul{\mc{C}} \to \ul{S} = \Spec k, p)\) 是一条光滑的 \(1\)-标记曲线。
    \begin{enumerate}
        \item 设 \((\mc{C}/S, p)\) 被赋予典范对数结构（于是 \(S\) 没有对数结构）。设 \(M_{\mc{C}} = P\)。则任何穿孔 \(p^{\flat} \colon M_{\mc{C}} \hookrightarrow M_{\mc{C}^{\circ}} \subset P^{\on{gp}}\) 都必须是平凡的。原因是 \(M_{\mc{C}^{\circ}}\) 不能有一个满足 \(\alpha_{\mc{C}^{\circ}}(s) = 0\) 的截面 \(s\)。若 \(s \in P = M_{\mc{C}}\)，则 \(\alpha_{\mc{C}^{\circ}}(s) = \alpha_{\mc{C}}(s)\)。否则，若 \(s \notin P\)，则因为 \(\alpha_{\mc{C}^{\circ}}(-s)\) 非零，\(\alpha_{\mc{C}^{\circ}}(s)\) 也非零。
        \item 设 \(M_{\mc{C}^{\circ}} \subseteq (\N \oplus \mc{O}_{\mc{C}}^{\times}) \oplus P^{\on{gp}}\) 是一个穿孔。选取 \(\mc{O}(-p)\) 的一个局部生成元 \(x\)。则任意 \(s \in M_{\mc{C}^{\circ}, p}\) 都具有形式 \(s = ((n, \varphi), x^m)\)，其中 \(n \in \N\) 且 \(\varphi \in \mc{O}_{\mc{C}, p}^{\times}\)。若 \(m < 0\)，则我们要求 \(\alpha_{\mc{C}^{\circ}}(s) = 0\)。这意味着 \(\alpha_{\mc{C}}((n, \varphi), 0) = 0\)，因此
            \[ \bar{M}_{\mc{C}^{\circ}, p} \subseteq \ab\{(n, m) \in \N \oplus \Z \mid m \geq 0 \text{ if } n = 0 \}. \]
            事实上，右端中任何包含 \(\N \oplus \N\) 的 fine 子幺半群都可以由某个穿孔实现。
        \item 设 \(S = \Spec(\N \to k)\)。考虑满足 \(\bar{M}_{\mc{C}^{\circ}, p} = \N^2 + \N (2,-2)\) 的 \(M_{\mc{C}^{\circ}}\)。因为 \((1,-1) \notin \bar{M}_{\mc{C}^{\circ}, p}\)，我们看到 \(M_{\mc{C}^{\circ}}\) 不是饱和的。
    \end{enumerate}
\end{exm}

\begin{lem}[{\cite{ACGS2025}}]
    设 \(Q\) 是一个尖 fs 幺半群。令 \(w \coloneqq \Hom(Q, \R_{\geq 0}) = Q_{\R}^{\vee}\)。设 \(Q^{\circ} \subseteq Q \oplus \Z\) 是有限生成的，满足 \(Q \oplus \N \subsetneq Q^{\circ}\) 以及 \(Q^{\circ} \cap \ab\{0\} \times \Z_{<0} = \emptyset\)。则存在一个非零的、凹的、分段线性函数
    \[ \ell \colon w \to \R_{\geq 0} \]
    具有有理斜率，使得
    \[ (Q^{\circ})_{\R}^{\vee} = \ab\{ (s, \lambda) \in w \times \R_{\geq 0} \mid 0 \leq \lambda \leq \ell(s) \}. \]
\end{lem}

\begin{exm}
    设 \(Q = \N^2\) 并令
    \begin{align*}
        Q^{\circ} &= \ab<((1,0), -1), ((0,1), -1), ((0,0),1)> \subset \N^2 \oplus \Z \\
        &= \ab\{((a,b), m) \mid a+b+m \geq 0, a,b \geq 0 \}.
    \end{align*}
    则
    \[ (Q^{\circ})_{\R}^{\vee} = \ab\{ (e_x, e_y, e_t) \mid 0 \leq e_t \leq \min(e_x, e_y) \}. \]
\end{exm}
