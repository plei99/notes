\section{\texorpdfstring{局部 \(\P^1\) 的高亏格 GW 理论的结构}{局部 P1 的高亏格 GW 理论的结构} (Shuai Guo)}%
\label{sec:Structure of higher-genus GW theory of local P1 (Shuai Guo)}

本讲基于与 Jingyi Xu 和 Qingsheng Zhang 的合作工作。

\subsection{\(X_p\) 的几何}%
\label{sub:The geometry of Xp}

我们考虑全空间
\[ X_p \coloneqq \on{Tot}\ab(\mc{O}_{\P^1}(p-1) \oplus \mc{O}_{\P^1}(-p-1)), \]
它是一个非紧的环面 Calabi-Yau 三维流形。当 \(p = 0\) 时，它就是解消 conifold。

设 \(N \simeq \Z^3\) 为秩 \(3\) 的格，设 \(M \coloneqq \Hom(N,\Z)\) 为其对偶格，我们将其等同于 \(\T \simeq (\C^{\times})^3\) 的特征标。\(X_p\) 的扇的一维锥由下式生成
\[ b_1 = (1-p,-1,1), \qquad b_2 = (0,1,1), \qquad b_3 = (1,0,1), \qquad b_4 = (0,0,1), \]
且 \(X_p\) 包含 \(\T\) 作为开稠密子集。利用对偶向量 \(e_3^{\vee} \in M\)，我们得到\textit{Calabi-Yau 子环面}
\[ \T' \coloneqq \ker\ab(\chi^{e_3^{\vee}} \colon \T \to \C^{\times}) \simeq (\C^{\times})^2, \]
其作用保持 Calabi-Yau 形式不变。环面图由两个三角形组成，分别对应于环面作用的两个不动点。在第一个不动点处 \(\T'\) 作用的权为
\[ (0,1), \qquad (1,0), \qquad (-1,-1), \]
而在第二个不动点处其权为
\[ (0,-1), \qquad (1,1-p), \qquad (-1,p). \]
因此我们可以通过在这两个不动点处作局部化，用等变上同调来表达不变量。这里，\(H^*_{\T'}(\mr{pt}) = \C[\lambda_1,\lambda_2]\)。

遗憾的是，当 \(p \geq 2\) 时 \(X_p\) 的环面图不是凸的。此时 \(X_p\) 不是半射影的。正如我们将在~\Cref{sub:Topological recursion and the remodeling conjecture} 中解释的，这正是当前情形未被已有文献覆盖的原因。

\subsection{CGMPS 猜想}%
\label{sub:The conjecture of CGMPS}

2007 年，Caporaso、Griguolo、Mari\~{n}o、Pasquetti 和 Seminara 研究了 \(X_p\) 的 Gromov-Witten 理论。记
\[ \check{F}^{X_p}(w,\hbar) \coloneqq \sum_{g \geq 0} \check{F}_g^{X_p}(w) \hbar^{2g-2} \]
为总自由能，他们精确计算了亏格一的部分，并提出了如下假设
\begin{equation}\label{eqn:cgmps ansatz}
    \check{F}_g^{X_p} = \frac{\wt{\mc{P}}_g(w,p)}{(w-w_c)^{5g-5}}, \qquad \wt{\mc{P}}_g(w,p) = \sum_{i=1}^{5g-5} a_{g,i}(p)(w-1)^i
\end{equation}
对 \(g \geq 2\) 成立。这里，变量 \(w\) 与 K\"ahler 参数 \(q\) 的关系为 \(w = 1-q\)，且 \(w_c = 1 - \frac{1}{p^2}\)。我们的约定将相差一个符号，即当 \(g \geq 2\) 时 \(F_g^{X_p} = (-1)^{g-1}\check{F}_g^{X_p}\)。

通过分析亏格零不变量的渐近行为，CGMPS 定义了标度化的弦耦合
\[ z^{5/2} = \hbar^{-2} \frac{p^8}{4w_c^5}(w-w_c)^5 \]
并引入了\textit{双标度极限} \(F_{\mr{ds}}(z)\)，它是将 \(\check{F}^{X_p}(w,\hbar)\) 在固定 \(z\) 的条件下令 \(w \to w_c\) 且 \(\hbar \to 0\) 而得到的。结合他们的亏格零和亏格一结果与上述假设，可得
\[ F_{\mr{ds}}(z) = -\frac{4}{15}z^{5/2} - \frac{1}{48}\log z + \sum_{g \geq 2} b_g z^{-5(g-1)/2}, \]
其中 \(b_g = \ab(\frac{p^8}{4w_c^5})^{g-1}\wt{\mc{P}}_g(w_c,p)\)。

\begin{conj}[CGMPS {\cite{CGMPS2007}}]%
\label{conj:cgmps}
    双标度极限 \(F_{\mr{ds}}(z)\) 与 \((3,2)\) 极小模型的自由能 \(F_{(3,2)}(z)\) 重合。
\end{conj}

\((3,2)\) 极小模型的自由形如
\[ F_{(3,2)}(z) = -\frac{4}{15}z^{5/2} - \frac{1}{48}\log z + \sum_{g\geq2} a_g z^{-5(g-1)/2}, \quad a_g = \frac{4^{1-g}}{(3g-3)!}\ab<\tau_2^{3g-3}>_g, \]
其中
\[ \ab<\tau_2^{3g-3}>_g \coloneqq \int_{\Mbar_{g,3g-3}} \psi_1^2 \cdots \psi_{3g-3}^2, \]
由此可见~\Cref{conj:cgmps} 等价于断言对一切 \(g \geq 2\) 都有 \(b_g = a_g\)。

\subsection{主要结果}%
\label{sub:The main results (Guo local P1)}

~\Cref{sec:Structure of higher-genus GW invariants of the quintic (Shuai Guo)} 中的整体镜像对称图像在此处同样适用。我们的目标是描述一维 B 模型模空间的三个边界点处 \(F_g^{X_p}\) 的行为。极限 \(q \to 0\) 为大复结构极限，给出以 A 模型变量 \(Q\) 表示的 Gromov-Witten 展开。在 \(q \to 1\) 处，我们将利用加框 \(R\) 矩阵证明 \(F_g\) 保持正则。有趣的点是 \(q \to \frac{1}{p^2}\)，它是一个高阶临界点。与普通的 conifold 奇点不同——其亏格 \(g\) 主极点的阶为 \(2g-2\)——该退化的极点阶至多为 \(5g-5\)。其首项系数由 \((3,2)\) 极小模型控制。

\begin{thm}%
\label{thm:local P1 mirror symmetry}
    对任意整数 \(p \geq 2\) 和任意 \(g \geq 2\)，\(X_p\) 的 Gromov-Witten 势由谱曲线 \(\mc{C}\) 的 Eynard-Orantin 拓扑递推计算，即
    \[ F_g^{X_p}(Q) = (-1)^{g-1}\mc{F}_g^{X_p}(q). \]
\end{thm}

特别地，这给出了重塑猜想的一个非半射影环面 Calabi-Yau 例子。

\begin{thm}%
\label{thm:local P1 rationality}
    当 \(g \geq 2\) 时，势 \(F_g^{X_p}\) 是一个有理函数的展开
    \[ F_g^{X_p} = (-1)^{g-1}\frac{\mc{P}_g(\Delta,p^2)}{\Delta^{5g-5}}, \qquad \Delta \coloneqq \frac{1-p^2q}{1-q} = \frac{p^2(w-w_c)}{w}, \]
    其中 \(\mc{P}_g \in \frac{1}{(p^2-1)^{2g-2}}\Q[p^2][\Delta]_{\leq 5g-5}\)。
\end{thm}

特别地，这证明了结构性假设~\Cref{eqn:cgmps ansatz}。

为了确定双标度极限，剩下需要确定 \(\mc{P}_g\) 在 \(\Delta = 0\) 处的常数项。

\begin{thm}%
\label{thm:local P1 constant term}
    \(\mc{P}_g\) 的常数项为
    \[ \mc{P}_g(0,p^2) = \frac{p^{6g-6}}{(p^2-1)^{2g-2}}\frac{1}{(3g-3)!}\ab<\tau_2^{3g-3}>_g. \]
\end{thm}

因为 \(\wt{\mc{P}}_g(w_c,p) = \mc{P}_g(0,p^2)\ab(\frac{1}{p^2}-\frac{1}{p^4})^{5g-5}\)，\Cref{thm:local P1 constant term} 给出
\[ b_g = \frac{4^{1-g}}{(3g-3)!}\ab<\tau_2^{3g-3}>_g = a_g, \]
这就证明了~\Cref{conj:cgmps}。

\subsection{\(X_p\) 的 Gromov-Witten 理论}%
\label{sub:Gromov-Witten theory of Xp}

关于局部曲线的等变理论及其局部化公式，见~\cite{BryanPandharipande2008}。

因为 \(X_p\) 非紧，模空间 \(\Mbar_{g,0}(X_p,d)\) 不是紧的，虚类的次数先验地没有定义。为了定义不变量，我们将引入环面作用并过渡到 \(\P^1\) 上的扭曲理论，在那里可以应用虚局部化。

我们选取一个加框 \(f \in \Z\)，并令 \(\T_f \coloneqq \ker(\lambda_2 - f\lambda_1)\)。两行局部化权，每行各含一个零权，为
\[ \begin{bmatrix} 0 & f & 1 & -f-1 \end{bmatrix}, \qquad \begin{bmatrix} -f & 0 & 1-fp+f & -1+fp \end{bmatrix}. \]
若 \(f \neq 0\)，则重新标度特征标可得到等价的对称权向量
\[ \begin{bmatrix} 1 & -1 & -\mf{f} & \mf{f} \end{bmatrix}, \qquad \mf{f} = 1-p+\frac{2}{f}, \]
我们将调整后的环面记为 \(\T_{\mf{f}}\)。若 \(H^*_{\T_{\mf{f}}}(\mr{pt}) = \C[\lambda]\)，则此规范化为
\[ \lambda_1 = -\frac{2}{\mf{f}}\lambda, \qquad \lambda_2 = -2\lambda. \]

设 \(\pi \colon \Mbar_{g,1}(\P^1,d) \to \Mbar_{g,0}(\P^1,d)\) 为遗忘映射，\(\ev\) 为在标记点处的取值映射。于是我们定义
\[ N^{X_p}_{g,d} \coloneqq \int_{[\Mbar_{g,0}(\P^1,d)]^{\vir}_{\T_{\mf{f}}}} e_{\T_{\mf{f}}}^{-1}\ab(R^{\bullet}\pi_* \ev^*(\mc{O}(p-1)\oplus\mc{O}(-p-1))), \]
其中 \(e_{\T_{\mf{f}}}\) 为等变 Euler 类。当 \(p=0\) 时，此式退化为通常的解消 conifold 公式。

我们得到一个 CohFT \(\Omega^{\lambda,\mf{f}}\)，其状态空间为等变上同调环
\[ H^*_{\T_{\mf{f}}}(\P^1) = \frac{\C[H,\lambda]}{(H+\lambda)(H-\lambda)} \]
配对为
\[ (a,b) \coloneqq \int_{\P^1} a \cup b \cup e^{-1}_{\T_{\mf{f}}}(\mc{O}(p-1)\oplus\mc{O}(-p-1)). \]
在平坦基 \(\ab\{\1,H\}\) 下 Gram 矩阵为
\[ \frac{1}{((p-\mf{f})^2-1)((p+\mf{f})^2-1)}\begin{pmatrix} 2\mf{f}p\lambda^{-3} & (1-\mf{f}^2-p^2)\lambda^{-2} \\ (1-\mf{f}^2-p^2)\lambda^{-2} & 2\mf{f}p\lambda^{-1} \end{pmatrix}. \]
我们定义关联函数为
\[ \Omega^{\lambda,\mf{f}}_{g,n}(\alpha_1,\ldots,\alpha_n) \coloneqq \sum_{d\geq0}\st_*\ab(\prod_{i=1}^n \ev_i^*\alpha_i \cap [\Mbar_{g,n}(\P^1,d)]^{\vir,\lambda,\mf{f}})e^{-dt}, \]
其中 \(\st\) 为到 \(\Mbar_{g,n}\) 的遗忘映射。扭曲虚类为
\[ [\Mbar_{g,n}(\P^1,d)]^{\vir,\lambda,\mf{f}} \coloneqq e^{-1}_{\T_{\mf{f}}}\ab(R^{\bullet}\pi_*\ev^*(\mc{O}(p-1)\oplus\mc{O}(-p-1))) \cap [\Mbar_{g,n}(\P^1,d)]^{\vir}_{\T_{\mf{f}}}. \]
记 \(Q = (-1)^{p+1}e^{-t}\)。则亏格 \(g\) 势为
\[ F^{X_p}_g(t) = \int_{\Mbar_g} \Omega^{\lambda,\mf{f}}_{g,0}. \]

\begin{rmk}
    正次数不变量 \(N^{X_p}_{g,d\geq1}\) 由拓扑顶点~\cite{LLLZ2009} 可知与 \(\mf{f}\) 无关，其 \(\lambda\) 无关性也可由~\Cref{thm:local P1 reconstruction} 得到。我们将在~\Cref{thm:local P1 rationality} 的证明中利用这种 \(\mf{f}\) 无关性。
\end{rmk}

在亏格零时，我们有
\[ N^{X_p}_{0,d} = \frac{(-1)^{(p+1)d-1}}{d^3}\binom{p^2d-1}{d-1}, \qquad d \geq 1, \]
由此可以精确计算 Yukawa 耦合：
\[ Y = -\partial_t^3 F_0^{X_p}(t) = \frac{1-\mf{f}^2-p^2}{((p-\mf{f})^2-1)((p+\mf{f})^2-1)} + \frac{q}{p^2q-1}. \]
这里，\(q\) 与 \(Q\) 通过镜像映射 \(Q = q(1-q)^{p^2-1}\) 相关联。利用 \((v_1 \star v_2, v_3) = \Omega^{\lambda,\mf{f}}_{0,3}(v_1,v_2,v_3)\) 计算量子积，得到
\[ H \star H = -\frac{(\mf{f}^2-1)q+1}{p^2q-1}\lambda^2 \1 - \frac{2\mf{f}pq}{p^2q-1}\lambda H, \]
或写成矩阵形式
\[ H \star = \begin{pmatrix} 0 & -\frac{(\mf{f}^2-1)q+1}{p^2q-1}\lambda^2 \\ 1 & -\frac{2\mf{f}pq}{p^2q-1}\lambda \end{pmatrix}. \]
该矩阵的特征值 \(L_{\alpha}\) 一般互不相同。于是 \(\Omega^{\lambda,\mf{f}}\) 是半单的。我们记 \(\ab\{e_{\alpha}\}\) 为幂等基，\(\Delta_{\alpha}^{-1} = (e_{\alpha},e_{\alpha})\) 为其长度，\(\bar{e}_{\alpha} \coloneqq \Delta_{\alpha}^{1/2}e_{\alpha}\) 为规范典范基。

因为该理论是半单的，\Cref{thm:teleman} 给出
\[ \Omega^{\lambda,\mf{f}} = R.T_R.\ab(\bigoplus_{\beta=0}^1 \Omega^{\mr{KW}}_{\beta}), \]
其中 \(\Omega^{\mr{KW}}_{\beta}\) 是一维平凡 CohFT，且 \(T_R(z) = z(\wt{\1}-R^{-1}(z)\1)\)，\(\wt{\1} \coloneqq \sum_{\beta}\bar{e}_{\beta}\)。我们由量子微分方程确定 \(R\) 矩阵
\[ z \d{S_{\tau}(z)} = \d{\tau} \star_{\tau} S_{\tau}(z). \]
等变局部化与量子 Riemann-Roch 将基本解与 \(R\) 矩阵联系起来：
\[ (S^*_{\tau}(z)\1, \bar{e}_{\alpha})\big|_{q=0}\, C_{\alpha}(z) = (\1, R(z)\bar{e}_{\alpha})e^{u^{\alpha}(q)/z}, \]
其中初值条件 \(C_{\alpha}(z)\) 是涉及 Bernoulli 数的指数函数。记 \(R_{i\bar{\alpha}}(z) \coloneqq (R^*(z)H^i, \bar{e}_{\alpha})\)，由量子微分方程得到二阶方程
\[ \ab( (zD+L_{\alpha})^2 + \frac{2\mf{f}pq}{p^2q-1}\lambda(zD+L_{\alpha}) + \frac{(\mf{f}^2-1)q+1}{p^2q-1}\lambda^2 )R_{0\bar{\alpha}} = 0, \]
其中 \(D = -\partial_t\)。然后我们可以逐阶计算 \(R\) 矩阵在 \(z\) 中的展开。

\subsection{拓扑递推与重塑猜想}%
\label{sub:Topological recursion and the remodeling conjecture}

我们现在转到 B 模型一侧。我们的目标是将拓扑递推给出的重构与上述 Givental-Teleman 重构等同起来。Eynard--Orantin 拓扑递推~\cite{EO2007} 的输入是一条\textit{谱曲线} \((\Sigma,x,y,B)\)，其中 \(\Sigma\) 为 Riemann 曲面；\(x\) 为局部定义的函数，其亚纯微分 \(\d{x}\) 有有限多个单零点；\(y\) 在这些零点附近全纯，且在那里 \(\d{y} \neq 0\)；\(B\) 为 \(\Sigma \times \Sigma\) 上的基本双微分。在 \(\d{x}\) 的零点 \(z^{\beta}\) 附近，设 \(\bar{z} \neq z\) 为满足 \(x(\bar{z}) = x(z)\) 的局部共轭点，并定义递推核
\[ K_{\beta}(z_0,z) \coloneqq \frac{\int_{\bar{z}}^z B(z_0,z')}{2(y(z)-y(\bar{z}))\d{x}(z)}. \]
若设 \(\omega_{0,0} = \omega_{1,0} = 0\)，\(\omega_{0,1}(z) = y(z)\d{x}(z)\)，\(\omega_{0,2}(z_1,z_2) = B(z_1,z_2)\)，则对 \(2g-2+n+1 > 0\) 递归地定义多重微分
\begin{align*}
    \omega_{g,n+1}(z_0,z_{[n]}) ={}& \sum_{\beta}\on{Res}_{z = z^{\beta}} K_{\beta}(z_0,z) \\
    &\cdot \Bigg( \omega_{g-1,n+2}(z,\bar{z},z_{[n]}) \\
    &\qquad + \sideset{}{'}\sum_{\substack{g_1+g_2 = g \\ I \sqcup J = [n]}}\omega_{g_1,\abs{I}+1}(z,z_I)\omega_{g_2,\abs{J}+1}(\bar{z},z_J)\Bigg),
\end{align*}
其中带撇求和不含 \(\omega_{0,1}\)。当 \(g \geq 2\) 时自由能为
\[ \mc{F}_g = \omega_{g,0} \coloneqq \frac{1}{2-2g}\sum_{\beta}\on{Res}_{z=z^{\beta}}\omega_{g,1}(z)\int_{z'=z^{\beta}}^z \omega_{0,1}(z'). \]

Bouchard、Klemm、Mari\~{n}o 和 Pasquetti 提出的重塑猜想~\cite{BKMP2009} 断言，环面 Calabi-Yau 三维流形 \(X\) 的全亏格开闭 Gromov-Witten 不变量，与其镜像曲线上拓扑递推产生的 \(\omega_{g,n}\) 不变量重合。该猜想对 \(\C^3\)、解消 conifold 和局部曲线已由 Eynard-Orantin、Zhou 和 Chen-Li-Liu 证明。随后 Fang--Liu--Zong~\cite{FLZ2020} 对一切半射影环面 Calabi-Yau 三维流形证明了该猜想。遗憾的是，当 \(p \geq 2\) 时 \(X_p\) 不是半射影的，因此落在其证明范围之外。

为了比较两侧，我们使用 Dunin-Barkowski、Orantin、Shadrin 和 Spitz~\cite{DOSS2014} 的重构，它将稳定的拓扑递推关联函数与半单 CohFT 的 Givental 重构等同起来。在每个临界点 \(z^{\beta}\) 周围，由下式定义局部 Airy 坐标 \(\eta^{\beta}(z)\)
\[ x(z) = x(z^{\beta}) + \frac{1}{2}(\eta^{\beta}(z))^2 \]
以及整体定义的亚纯参考形式
\[ \d{\zeta^{\bar{\beta}}}(z) \coloneqq -\on{Res}_{z'=z^{\beta}}\frac{B(z',z)}{\eta^{\beta}(z')}. \]
相应 CohFT 的状态空间 \(V\) 由对应于临界点的基 \(\ab\{\bar{e}_{\beta}\}\) 张成，配对为对角形 \(\eta(\bar{e}_{\beta},\bar{e}_{\gamma}) = \delta_{\beta,\gamma}\)。曲线 \(R\) 矩阵 \(\check{R}\) 由渐近 Laplace 变换定义
\[ \frac{u}{\sqrt{2\pi u}}\int_{\mc{L}_{\gamma}} e^{-x(z)/u}\d{\zeta^{\bar{\beta}}}(z) \asymp e^{-x^{\gamma}/u}\eta\ab(\bar{e}_{\gamma}, \check{R}^*(-u)\bar{e}_{\beta}), \]
而对 \(\d{y}\) 的类似变换定义 \(T(u)\)，若所得 CohFT 具有平坦单位，则 \(T(u) = u(\wt{\1}-\check{R}^{-1}(u)\1)\)。于是 \(\check{R}\) 是辛的，曲线 CohFT 为 \(\check{\Omega} = \check{R}.T.\ab(\bigoplus_{\beta}\Omega^{\mr{KW}}_{\beta})\)，记 \(\d{\zeta^{\bar{\beta}}_k} \coloneqq \ab(-\d{} \circ \frac{1}{\d{x}})^k \d{\zeta^{\bar{\beta}}}\)，对应关系为
\begin{align*}
    \omega_{g,n}(z_1,\ldots,z_n) ={}& \sum_{\substack{k_1,\ldots,k_n \geq 0 \\ \beta_1,\ldots,\beta_n \in [N]}}\ab(\int_{\Mbar_{g,n}} \check{\Omega}_{g,n}(\bar{e}_{\beta_1}\psi_1^{k_1},\ldots,\bar{e}_{\beta_n}\psi_n^{k_n})) \\
    &\cdot \prod_{i=1}^n \d{\zeta^{\beta_i}_{k_i}}(z_i).
\end{align*}

对 \(X_p\)，我们考虑谱曲线
\[ \mc{C}_{\mf{f}} = \ab(\P^1, x_{\mf{f}}(z), y(z), B(z_1,z_2) = \frac{\d{z_1}\d{z_2}}{(z_1-z_2)^2}), \qquad x_{\mf{f}}(z) = x(z) + 2\mf{f}\lambda^2 y(z), \]
其中
\begin{align*}
    x(z) &= t^0 + \lambda \log q + \lambda(p-1)\log z + \lambda(-p-1)\log\frac{q-z}{1-z}, \\
    y(z) &= \frac{p+1}{2\lambda}\log(1-q) + \frac{1}{2\lambda}\log z - \frac{1}{2\lambda}\log(1-z) - \frac{1}{2\lambda}\log(q-z).
\end{align*}
解 \(\d{x_{\mf{f}}} = 0\)，得到临界点
\[ z^{\alpha} = \frac{1-\ms{L}_{\alpha}}{1+\mf{f}+p\ms{L}_{\alpha}}, \qquad \ms{L}_{\alpha} = \lambda^{-1}L_{\alpha}, \qquad q = \frac{\ms{L}_{\alpha}^2-1}{(p\ms{L}_{\alpha}+1)^2-1}, \]
内积为 \(\eta(e_{\alpha},e_{\beta}) = \delta_{\alpha,\beta}\frac{y'(z^{\alpha})^2}{x''_{\mf{f}}(z^{\alpha})}\)。于是所得长度满足 \(\check{\Delta}_{\alpha} = \Delta_{\alpha}\)。因此，谱曲线的 Frobenius 结构就是 \(X_p\) 的量子积。

至此我们已匹配了 Frobenius 结构，剩下需要匹配两个 \(R\) 矩阵。在 Lefschetz thimble 上分部积分得到
\begin{align*}
    u \partial_{t^0}\ab(e^{-x^{\gamma}/u}\check{R}^*(-u)^{\bar{\gamma}}_{\1}) &= -e^{-x^{\gamma}/u}\check{R}^*(-u)^{\bar{\gamma}}_H, \\
    u\partial_{t^0}\ab(e^{-x^{\gamma}/u}\check{R}^*(-u)^{\bar{\gamma}}_H) &= -\lambda b(q)e^{-x^{\gamma}/u}\check{R}^*(-u)^{\bar{\gamma}}_H - \lambda^2 c(q) e^{-x^{\gamma}/u}\check{R}^*(-u)^{\bar{\gamma}}_{\1},
\end{align*}
其中
\[ c(q) = -\frac{(\mf{f}^2-1)q+1}{p^2q-1}, \qquad b(q) = -\frac{2\mf{f}pq}{p^2q-1} \]
恰为上面算出的量子积的系数，因此它们与 Gromov-Witten \(R\) 矩阵的量子微分方程一致。为了匹配初值条件，我们取 Laplace 变换沿 thimble 在 \(q \to 0\) 时的极限，这给出 Beta 函数从而给出 Gamma 函数之积。然后 Stirling 展开与 Bernoulli 数初值条件 \(C_{\alpha}(z)\) 相匹配。

\begin{prop}%
\label{prop:R matrices coincide}
    两个 \(R\) 矩阵重合，即 \(R(z) = \check{R}(u)|_{u=-z}\)。
\end{prop}

至此我们已证明典范长度与 \(R\) 矩阵均一致。作代换 \(u = -z\) 后，亏格 \(g\) 稳定图求和相差整体奇偶因子 \((-1)^{3g-3} = (-1)^{g-1}\)，这就证明了~\Cref{thm:local P1 mirror symmetry}。

\subsection{有理性}%
\label{sub:Rationality}

因为高亏格势与 \(\mf{f}\) 无关，为简单起见可设 \(\mf{f} = 0\)。这同时简化了典范基和量子微分方程，后者变为
\[ \ab((zD+L_{\alpha})^2 - \frac{1-q}{1-p^2q}\lambda^2)R_{0\bar{\alpha}} = 0, \qquad L_{\alpha} = (-1)^{\alpha}\lambda L, \quad L = \ab(\frac{1-q}{1-p^2q})^{1/2}, \]
我们将 \(R\) 矩阵的矩阵元记为 \((R_k)_i^{\alpha}\)。设
\[ \Delta \coloneqq \frac{1}{L^2} = \frac{1-p^2q}{1-q}, \qquad \mc{R} \coloneqq \Q[p^2,\Delta], \qquad \mc{R}_{\leq d} \coloneqq \ab\{P \in \mc{R} \mid \deg_{\Delta} P \leq d\}. \]
我们现在对量子微分方程的系数作归纳，得到多项式结构。

\begin{prop}%
\label{prop:R matrix polynomiality}
    对 \(k \in \Z_{\geq0}\) 和 \(i = 0,1\)，有
    \[ (R_k)_i^{\alpha} \in \frac{1}{(p^2-1)^k \lambda^{k-i}\Delta^{(3k+i)/2}}\mc{R}_{\leq 2k}. \]
\end{prop}

分析 Givental-Teleman 重构中稳定图求和的次数贡献，并追踪 \(\lambda\)、\(\Delta\) 和 \(p^2-1\) 的幂次，得到如下结果。

\begin{thm}%
\label{thm:local P1 reconstruction}
    记 \([\,\cdot\,]_d\) 为余维 \(d\) 分量，则有
    \begin{align*}
        \ab[\Omega^{\lambda}_{g,n}(H^{a_1},\ldots,H^{a_n})]_d \in{}& \frac{\lambda^{3g-3-d+\sum_i a_i}}{(p^2-1)^{d-(g-1)}\Delta^{\frac{1}{2}(g-1+3d+\sum_i a_i)}}\mc{R}_{\leq 2d} \\
        &\otimes H^{2d}(\Mbar_{g,n},\Q).
    \end{align*}
\end{thm}

在~\Cref{thm:local P1 reconstruction} 中取 \(n=0\) 且 \(d = 3g-3 = \dim \Mbar_g\)，得到
\[ F_g^{X_p} \in \frac{1}{(p^2-1)^{2g-2}\Delta^{5g-5}}\mc{R}_{\leq 6g-6}, \]
因此若分子在 \(\Delta\) 中的次数不超过 \(5g-5\)，则~\Cref{thm:local P1 rationality} 得证。我们可以在 \(q=1\) 处看到这一界。因为 \(\Delta\) 在那里有单极点，\(F_g^{X_p}\) 的正则性迫使分子消去 \(\Delta\) 的最高次幂。于是我们把有理性约化为在 \(q=1\) 处的正则性。该正则性由 \(F_g^{X_p}\) 的 \(\mf{f}\) 无关性以及当 \(\mf{f} \neq 0\) 时 \(R\) 矩阵在 \(q \to 1\) 时极限的存在性得到。

\subsection{\(\Delta = 0\) 的极限}%
\label{sub:The limit Delta = 0}

为了确定剩下的常数项，我们现在研究极限 \(\Delta = 0\)，它恰为双标度极限。我们将典范长度重新标度为
\[ \delta_{\alpha} \coloneqq \Delta^{1/2}\Delta_{\alpha} = (-1)^{\alpha+1}2(p^2-1)\lambda^3 \]
并通过分离~\Cref{prop:R matrix polynomiality} 中在 \(\Delta = 0\) 处的主项来定义修正的 \(R\) 矩阵元
\[ (\bar{R}_k)_i^{\alpha} \coloneqq \ab[(-1)^k (R_k)_i^{\alpha}\Delta^{(3k+i)/2}]\Big|_{\Delta=0}, \qquad i = 0,1. \]
于是 \(F_g^{X_p}\) 的图求和在把 \(\Delta_{\alpha}\) 换成 \(\delta_{\alpha}\)、\((R_k)_i^{\alpha}\) 换成 \((\bar{R}_k)_i^{\alpha}\) 之后，就成为 \(\mc{P}_g(0,p^2)\) 的公式。

\begin{lem}%
\label{lem:R matrix at Delta zero}
    解 \(\Delta = 0\) 处的主阶量子微分方程，得到闭式
    \begin{align*}
        (\bar{R}_k)_0^{\alpha} &= \frac{p^{2k}}{(p^2-1)^k\lambda^k}(-1)^{k\alpha}\prod_{n=1}^k \frac{(6n-7)(6n+1)}{48n}, \\
        (\bar{R}_k)_1^{\alpha} &= \frac{p^{2k}}{(p^2-1)^k \lambda^{k-1}}(-1)^{(k-1)\alpha}\frac{1-6k}{1+6k}\prod_{n=1}^k \frac{(6n-7)(6n+1)}{48n}.
    \end{align*}
\end{lem}

在~\Cref{lem:R matrix at Delta zero} 中提出 \(p\) 相关因子后，我们得到纯有理系数。我们现在在 \(V = \on{span}\ab\{\phi_0,\phi_1\}\) 上构造拓扑场论 \(\Omega^{\mr{top}}\)，其单位为 \(\phi_0\)，其中
\[ (\phi_0,\phi_0) = (\phi_1,\phi_1) = 0, \qquad (\phi_0,\phi_1) = -1, \qquad \phi_1 \star \phi_1 = \phi_0. \]
用 \(R(z)\) 及其平坦单位平移作重构，得到常数 \(C_g\) 满足
\[ \mc{P}_g(0,p^2) = \frac{p^{6g-6}}{(p^2-1)^{2g-2}}C_g, \]
剩下需要确定重构出的 CohFT。

\((2,3)\) 极小模型的谱曲线为
\[ x(z) = \frac{1}{3}(z^3-3\ep^2 z) + \tau^0, \qquad y(z) = \frac{1}{2}(z^2-2), \]
平坦坐标为 \(\tau^1 = \ep^4/2\)。若设 \(\tau^0 = 0\) 且 \(\ep = 1\)，并等同 \(\phi_0 = \partial_{\tau^0}\) 与 \(\phi_1 = \partial_{\tau^1}\)，则由 \(X_p\) 在 \(\Delta = 0\) 处得到的 \(R\) 矩阵与该曲线的 \(R\) 矩阵重合。因此，\(C_g\) 恰为 \((2,3)\) 极小模型的自由能。

最后，我们应用辛交换 \((x,y) \mapsto (y,-x)\)，在此之下曲线变为
\[ x(z) = \frac{1}{2}(z^2-2), \qquad y(z) = -\frac{1}{3}(z^3-3z). \]
交换后的曲线具有一维状态空间、平凡 \(R\) 矩阵和平移 \(T(u) = u^2 \bar{e}\)，从而
\[ C_g = \frac{1}{(3g-3)!}\ab<\tau_2^{3g-3}>_g, \]
这就是~\Cref{thm:local P1 constant term}。注意此交换是 \((2,3)\) 与 \((3,2)\) 极小模型之间的对应，这正是为什么此处出现的相交数与~\Cref{conj:cgmps} 中的相同。
